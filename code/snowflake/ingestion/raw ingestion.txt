---- VALIDATION 
SELECT 'customers' AS table_name, COUNT(*) AS row_count FROM ECOMMERCE.RAW.OLIST_CUSTOMERS
UNION ALL SELECT 'geolocation', COUNT(*) FROM ECOMMERCE.RAW.OLIST_GEOLOCATION
UNION ALL SELECT 'orders', COUNT(*) FROM ECOMMERCE.RAW.OLIST_ORDERS
UNION ALL SELECT 'order_items', COUNT(*) FROM ECOMMERCE.RAW.OLIST_ORDER_ITEMS
UNION ALL SELECT 'order_payments', COUNT(*) FROM ECOMMERCE.RAW.OLIST_ORDER_PAYMENTS
UNION ALL SELECT 'order_reviews', COUNT(*) FROM ECOMMERCE.RAW.OLIST_ORDER_REVIEWS
UNION ALL SELECT 'products', COUNT(*) FROM ECOMMERCE.RAW.OLIST_PRODUCTS
UNION ALL SELECT 'sellers', COUNT(*) FROM ECOMMERCE.RAW.OLIST_SELLERS
UNION ALL SELECT 'product_category_translation', COUNT(*) FROM ECOMMERCE.RAW.PRODUCT_CATEGORY_NAME_TRANSLATION;
